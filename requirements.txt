# Runtime dependencies for the Sietch CRM dashboard.
#
# CI (.github/workflows/smoke.yml) installs ONLY this file on a clean runner,
# so every third-party module imported at module scope by server.py, db.py,
# the scanner, or a test suite has to be listed here. The Dockerfile installs
# the same set explicitly; keep the two lists aligned, otherwise CI fails with
# ModuleNotFoundError before a single test runs (and the suites that "SKIP
# when server is not importable" quietly stop covering anything).

# PostgreSQL driver (db.py).
psycopg2-binary>=2.9

# HTTP client used by server.py, the OAuth providers, mail send and Graph.
httpx>=0.27

# IMAP client used by scanner/mail_scanner.py (imported at module scope).
imap-tools>=1.8

# PDF generation.
fpdf2>=2.7.0
# ReportLab powers the modern (Sherwood-style) Invoice + Certificate of
# Completion PDFs in the doc_pdf/ package. Keep the pin compatible with
# Python 3.12 (the Docker base image is python:3.12-slim-bookworm).
reportlab>=4.0,<5

# Imported lazily (inside request handlers), but part of the app: installed
# so a runtime path does not depend on the container's extra packages.
Pillow>=10
ExifRead>=3

# Test-only: the PDF text/page-count helpers in tests/test_invoice_coc_modern.py
# fall back to pypdf when poppler's pdftotext is not installed (poppler-utils
# is an OS package, not a Python dependency, so it is absent on a bare runner).
pypdf>=4.0
